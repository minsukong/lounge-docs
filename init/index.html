실제 코드 또는 명령어 예시가 들어갑니다.
'use client';
function requestLocation() {
  const bridgeWindow = window as any;
  const message = JSON.stringify({
    requestId: crypto.randomUUID(),
    method: 'location.getCurrentPosition',
    params: {},
  });

  // Android WebView
  if (bridgeWindow.AppBridge) {
    bridgeWindow.AppBridge.postMessage(message);
    return;
  }

  // iOS WebView
  bridgeWindow.webkit?.messageHandlers?.AppBridge?.postMessage(message);
}
